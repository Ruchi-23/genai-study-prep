// SAFE CONFIGURATION: No token in source code! const GIST_CONFIG = { gistId: "f98a8a31728e6d29e0fb22410dbe440c", filename: "master_os_progress.json" }; // Retrieve token from private localStorage, or prompt once if missing function getSecureToken() { let token = localStorage.getItem("study_os_gh_token"); if (!token) { token = prompt("Enter your GitHub Personal Access Token (stored locally in your browser only):"); if (token) { token = token.trim(); localStorage.setItem("study_os_gh_token", token); } } return token; } // Reset / Change Token anytime function resetToken() { localStorage.removeItem("study_os_gh_token"); alert("Stored token cleared from this browser."); } // PUSH TO GIST async function pushToGist(notify = true) { const token = getSecureToken(); if (!token) return alert("A GitHub token with 'gist' scope is required to save."); setSyncStatus("syncing", "Saving to GitHub..."); try { const payload = { files: { [GIST_CONFIG.filename]: { content: JSON.stringify(state, null, 2) } } }; const res = await fetch(`https://api.github.com/gists/${GIST_CONFIG.gistId}`, { method: "PATCH", headers: { "Authorization": `Bearer ${token}`, "Accept": "application/vnd.github.v3+json", "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (res.ok) { setSyncStatus("synced", "Cloud Synced"); if (notify) alert("Saved to GitHub Cloud successfully!"); } else { if (res.status === 401) { localStorage.removeItem("study_os_gh_token"); alert("Token invalid or expired. Please re-enter your new token."); } else { alert(`Cloud save failed (Status ${res.status}).`); } setSyncStatus("error", "Sync Error"); } } catch (err) { setSyncStatus("error", "Network Error"); if (notify) alert("Network error during cloud sync."); } } // PULL FROM GIST async function pullFromGist(notify = true) { const token = getSecureToken(); if (!token) return; setSyncStatus("syncing", "Loading from Cloud..."); try { const res = await fetch(`https://api.github.com/gists/${GIST_CONFIG.gistId}`, { headers: { "Authorization": `Bearer ${token}`, "Accept": "application/vnd.github.v3+json" } }); if (res.ok) { const gistData = await res.json(); const fileData = gistData.files[GIST_CONFIG.filename]; if (fileData && fileData.content && fileData.content.trim() !== "{}" && fileData.content.trim() !== "") { state = JSON.parse(fileData.content); localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); applyTheme(); updateStats(); renderWeekPills(); renderDays(); setSyncStatus("synced", "Cloud Loaded"); if (notify) alert("Loaded your latest progress from GitHub Cloud!"); } else { setSyncStatus("synced", "Cloud Ready (Empty)"); if (notify) alert("Gist is empty. Click Cloud Save to upload your initial baseline."); } } else { setSyncStatus("error", "Load Error"); if (notify) alert("Cloud load failed."); } } catch (err) { setSyncStatus("error", "Offline"); if (notify) alert("Could not connect to GitHub Cloud."); } }
